    users {
        integer id PK
        text email UK
        text password_hash
        text name
        text role "customer | admin"
        integer created_at
        integer updated_at
    }
    sessions {
        text id PK "sha256(token)"
        integer user_id FK "cascade"
        integer expires_at
        integer created_at
    }
    password_reset_tokens {
        text id PK "sha256(token)"
        integer user_id FK "cascade"
        integer expires_at
        integer used_at "null = unused"
        integer created_at
    }
    addresses {
        integer id PK
        integer user_id FK "cascade"
        text name
        text line1
        text line2 "nullable"
        text city
        text state
        text postcode
        text country "default AU"
        text phone "nullable"
        boolean is_default
        integer created_at
    }
    categories {
        integer id PK
        text name
        text slug UK
        integer sort_order
        integer created_at
    }
    products {
        integer id PK
        text slug UK
        text name
        text subtitle "nullable"
        text description
        integer category_id FK "set null"
        boolean active
        boolean featured
        integer created_at
        integer updated_at
    }
    variants {
        integer id PK
        integer product_id FK "cascade"
        text sku UK
        text label "e.g. 10 mg"
        integer price_cents
        integer compare_at_cents "nullable, sale display"
        integer stock
        json attributes "Record<string,string>"
        boolean active
        integer sort_order
        integer created_at
        integer updated_at
    }
    product_images {
        integer id PK
        integer product_id FK "cascade"
        text r2_key "file lives in R2"
        text alt
        integer sort_order
        integer created_at
    }
    orders {
        integer id PK
        integer user_id FK "set null, nullable"
        text email
        text status "pending | paid | fulfilled | cancelled | refunded"
        text currency "default aud"
        integer subtotal_cents
        integer shipping_cents
        integer gst_cents
        integer total_cents
        json shipping_address "ShippingAddress"
        text stripe_session_id UK
        text stripe_payment_intent_id
        integer paid_at "nullable"
        integer fulfilled_at "nullable"
        text tracking_number "nullable"
        integer created_at
        integer updated_at
    }
    order_items {
        integer id PK
        integer order_id FK "cascade"
        integer variant_id FK "set null, nullable"
        text product_name "snapshot"
        text variant_label "snapshot"
        text sku "snapshot"
        integer unit_price_cents "snapshot"
        integer quantity
    }
    stripe_events {
        text id PK "evt_..."
        text type
        integer processed_at
    }
    contact_messages {
        integer id PK
        text name
        text email
        text subject
        text message
        text status "new | read | archived"
        integer created_at
    }

        +Date updatedAt

    class UserRole {
        <<enumeration>>
        customer
        admin
    }
    class Session {
        +string id  sha256 of token
        +number userId
        +Date expiresAt
        +Date createdAt
    }
    class Address {
        +number id
        +number userId
        +string name
        +string line1
        +string? line2
        +string city
        +string state
        +string postcode
        +string country
        +string? phone
        +boolean isDefault
    }
    class Category {
        +number id
        +string name
        +string slug
        +number sortOrder
    }
    class Product {
        +number id
        +string slug
        +string name
        +string? subtitle
        +string description
        +number? categoryId
        +boolean active
        +boolean featured
    }
    class Variant {
        +number id
        +number productId
        +string sku
        +string label
        +number priceCents
        +number? compareAtCents
        +number stock
        +object attributes  JSON map
        +boolean active
        +number sortOrder
    }
    class ProductImage {
        +number id
        +number productId
        +string r2Key
        +string alt
        +number sortOrder
    }
    class Order {
        +number id
        +number? userId
        +string email
        +OrderStatus status
        +number subtotalCents
        +number shippingCents
        +number gstCents
        +number totalCents
        +ShippingAddress shippingAddress
        +string? stripeSessionId
        +Date? paidAt
        +Date? fulfilledAt
        +string? trackingNumber
    }
    class OrderStatus {
        <<enumeration>>
        pending
        paid
        fulfilled
        cancelled
        refunded
    }
    class ShippingAddress {
        <<value object in JSON column>>
        +string name
        +string line1
        +string? line2
        +string city
        +string state
        +string postcode
        +string country
        +string? phone
    }
    class OrderItem {
        +number id
        +number orderId
        +number? variantId
        +string productName
        +string variantLabel
        +string sku
        +number unitPriceCents
        +number quantity
    }
    class StripeEvent {
        +string id
        +string type
        +Date processedAt
    }
    class ContactMessage {
        +number id
        +string name
        +string email
        +string subject
        +string message
        +ContactStatus status
    }
    class ContactStatus {
        <<enumeration>>
        new
        read
        archived
    }
